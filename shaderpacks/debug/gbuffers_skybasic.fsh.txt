1914:     vec3 GetSky(float VdotU, float VdotS, float dither, bool doGlare, bool doGround) {
1915:         // Prepare variables
1916:         float nightFactorSqrt2 = sqrt2(nightFactor);
1917:         float nightFactorM = sqrt2(nightFactorSqrt2) * 0.4;
1918:         float VdotSM1 = pow2(max(VdotS, 0.0));
1919:         float VdotSM2 = pow2(VdotSM1);
1920:         float VdotSM3 = pow2(pow2(max(-VdotS, 0.0)));
1921:         float VdotSML = sunVisibility > 0.5 ? VdotS : -VdotS;
1922: 
1923:         float VdotUmax0 = max(VdotU, 0.0);
1924:         float VdotUmax0M = 1.0 - pow2(VdotUmax0);
1925: 
1926:         // Prepare colors
1927:         vec3 upColor = mix(nightUpSkyColor * (1.5 - 0.5 * nightFactorSqrt2 + nightFactorM * VdotSM3 * 1.5), dayUpSkyColor, sunFactor);
1928:         vec3 middleColor = mix(nightMiddleSkyColor * (3.0 - 2.0 * nightFactorSqrt2), dayMiddleSkyColor * (1.0 + VdotSM2 * 0.3), sunFactor);
1929:         vec3 downColor = mix(nightDownSkyColor, dayDownSkyColor, (sunFactor + sunVisibility) * 0.5);
1930: 
1931:         // Mix the colors
1932:             // Set sky gradient
1933:             float VdotUM1 = pow2(1.0 - VdotUmax0);
1934:                   VdotUM1 = pow(VdotUM1, (1.0 - VdotSM2 * 0.4) * (1.15 - 0.15 * rainFactor2));
1935:                   VdotUM1 = mix(VdotUM1, 1.0, rainFactor2 * 0.15);
1936:             vec3 finalSky = mix(upColor, middleColor, VdotUM1);
1937: 
1938:             // Add sunset color
1939:             float VdotUM2 = pow2(1.0 - abs(VdotU));
1940:                   VdotUM2 = VdotUM2 * VdotUM2 * (3.0 - 2.0 * VdotUM2);
1941:                   VdotUM2 *= (0.7 - nightFactorM + VdotSM1 * (0.3 + nightFactorM)) * invNoonFactor * sunFactor;
1942:             finalSky = mix(finalSky, sunsetDownSkyColorP * (1.0 + VdotSM1 * 0.3), VdotUM2 * invRainFactor);
1943: 
1944:             // Add sky ground with fake light scattering
1945:             float VdotUM3 = min(max0(-VdotU + 0.08) / 0.35, 1.0);
1946:                   VdotUM3 = smoothstep1(VdotUM3);
1947:             vec3 scatteredGroundMixer = vec3(VdotUM3 * VdotUM3, sqrt1(VdotUM3), sqrt3(VdotUM3));
1948:                  scatteredGroundMixer = mix(vec3(VdotUM3), scatteredGroundMixer, 0.75 - 0.5 * rainFactor);
1949:             finalSky = mix(finalSky, downColor, scatteredGroundMixer);
1950:         //
1951: 
1952:         // Sky Ground
1953:         if (doGround)
1954:             finalSky *= smoothstep1(pow2(1.0 + min(VdotU, 0.0)));
1955: 
1956:         // Apply Underwater Fog
1957:         if (isEyeInWater == 1)
1958:             finalSky = mix(finalSky * 3.0, waterFogColor, VdotUmax0M);
1959: 
1960:         // Sun/Moon Glare
1961:         if (doGlare) {
1962:             if (0.0 < VdotSML) {
1963:                 float glareScatter = 3.0 * (2.0 - clamp01(VdotS * 1000.0));
1964:                 #ifndef SUN_MOON_DURING_RAIN
1965:                     glareScatter *= 1.0 - 0.75 * rainFactor2;
1966:                 #endif
1967:                 float VdotSM4 = pow(abs(VdotS), glareScatter);
1968: 
1969:                 float visfactor = 0.075;
1970:                 float glare = visfactor / (1.0 - (1.0 - visfactor) * VdotSM4) - visfactor;
1971:                 glare *= 0.7;
1972: 
1973:                 float glareWaterFactor = isEyeInWater * sunVisibility;
1974:                 vec3 glareColor = mix(vec3(0.38, 0.4, 0.5) * 0.3, vec3(1.5, 0.7, 0.3) + vec3(0.0, 0.5, 0.5) * noonFactor, sunVisibility);
1975:                      glareColor = glareColor + glareWaterFactor * vec3(7.0);
1976: 
1977:                 #ifdef SUN_MOON_DURING_RAIN
1978:                     glare *= 1.0 - 0.6 * rainFactor;
1979:                 #else
1980:                     glare *= 1.0 - 0.8 * rainFactor;
1981:                 #endif
1982:                 #if RAIN_STYLE == 1
1983:                     float glareDesaturateFactor = 0.5 * rainFactor;
1984:                 #elif RAIN_STYLE == 2
1985:                     float glareDesaturateFactor = rainFactor;
1986:                 #endif
1987:                 glareColor = mix(glareColor, vec3(GetLuminance(glareColor)), glareDesaturateFactor);
1988: 
1989:                 finalSky += glareColor * glare * shadowTime;
1990:             }
1991:         }
1992: 
1993:         #ifdef CAVE_FOG
1994:             // Apply Cave Fog
1995:             finalSky = mix(finalSky, caveFogColor, GetCaveFactor() * VdotUmax0M);
1996:         #endif
1997: 
1998:         // Dither to fix banding
1999:         finalSky = max(finalSky + (dither - 0.5) / 128.0, vec3(0.0));
2000: 
2001:         return finalSky;
2002:     }
